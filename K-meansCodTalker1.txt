#Kmeans Clustering  
#Importing important libraries 
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
%matplotlib inline
#getting data from local machine
df = pd.read_excel("C:/Users/sata/Desktop/mydata")
df.head()
# dropping unwanted columns and retain the need ones   with their column names
df1= df.drop(df.columns[[0, 1, 2]], axis = 1, inplace = True)  

# Finding optimal clusters using Elbow Method
wcss = []
for i in range(1,11):
    kmeans = KMeans(n_clusters = i, init = "k-means++", 
                   max_iter = 300, n_init = 10)
    kmeans.fit(df1)
    wcss.append(kmeans.inertia_)
plt.plot(range(1,11), wcss)
plt.title("Elbow graph")
plt.xlabel("Number of clusters")
plt.ylabel("WCSS")
plt.show()



#Clustering Based on the optimal cluster number in our case, k = 3  

kmeans = KMeans(n_clusters =3, init= "k-means++", 
               max_iter = 300, n_init = 10)
labels = kmeans.fit_predict(df1)

# Mapping Observation with appropriate cluster

cluster_output = pd.concat([df1,pd.DataFrame(labels)], axis = 1)
# Exporting clustered data to my desired location 
cluster_output.to_excel("C:/Users/User/Desktop/clustered_data.xlsx")
